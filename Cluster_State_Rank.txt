```python
"""
calculate_cluster_state_rank.py

PURPOSE:
    Rank unique clusters within each state by Cluster_Suitability_Value
    descending. Tiebreaker is Cluster_Capacity_kWdc descending.

    Every polygon belonging to the same Cluster_ID receives the same
    Cluster_State_Rank.

RANKING LOGIC:
    - Grouped by State_Name
    - Unique Cluster_IDs are ranked
    - Primary sort   : Cluster_Suitability_Value descending
    - Tiebreaker     : Cluster_Capacity_kWdc descending
    - Rank 1 = highest suitability cluster in that state

OUTPUT:
    - Cluster_State_Rank field updated in place

NOTES:
    - Run in ArcGIS Pro notebook
    - Requires arcpy
    - Cluster_Suitability_Value and Cluster_Capacity_kWdc must already
      be populated
    - Polygons with null Cluster_ID receive null Cluster_State_Rank
"""

import arcpy
from collections import defaultdict
from datetime import datetime


# ── USER CONFIGURATION ────────────────────────────────────────────────────────

FEATURE_CLASS = r"path\to\your\geodatabase.gdb\your_feature_class"  # <-- UPDATE

# ─────────────────────────────────────────────────────────────────────────────


start_time = datetime.now()

print("=" * 60)
print("CALCULATE CLUSTER STATE RANK")
print(f"Started : {start_time.strftime('%Y-%m-%d %H:%M:%S')}")
print("=" * 60)


# ── VALIDATE FEATURE CLASS ────────────────────────────────────────────────────

print(f"Feature class : {FEATURE_CLASS}")

if not arcpy.Exists(FEATURE_CLASS):
    raise FileNotFoundError(f"Feature class not found: {FEATURE_CLASS}")

print("Feature class confirmed.")


# ── GET TOTAL ROW COUNT ───────────────────────────────────────────────────────

total_rows = int(arcpy.GetCount_management(FEATURE_CLASS).getOutput(0))
print(f"Total rows    : {total_rows:,}")


# ── STEP 1: BUILD UNIQUE CLUSTER LOOKUP PER STATE ─────────────────────────────

print("=" * 60)
print("STEP 1 - READING UNIQUE CLUSTERS PER STATE")
print("=" * 60)

# state_clusters[state] = {
#     cluster_id: (suitability, capacity)
# }

state_clusters = defaultdict(dict)

null_cluster_count = 0
skipped_nulls = 0

fields = [
    "Cluster_ID",
    "State_Name",
    "Cluster_Suitability_Value",
    "Cluster_Capacity_kWdc"
]

print(f"Reading fields: {fields}")


with arcpy.da.SearchCursor(FEATURE_CLASS, fields) as cursor:

    for row in cursor:

        cluster_id = row[0]
        state_name = row[1]
        suitability = row[2]
        capacity = row[3]

        # No cluster ID
        if cluster_id is None:
            null_cluster_count += 1
            continue

        # Missing ranking information
        if (
            state_name is None
            or suitability is None
            or capacity is None
        ):
            skipped_nulls += 1
            continue

        # Store only ONE record per unique Cluster_ID
        #
        # If multiple polygons belong to the same cluster,
        # they will only be stored once.
        if cluster_id not in state_clusters[state_name]:

            state_clusters[state_name][cluster_id] = (
                suitability,
                capacity
            )


print("Read complete.")

print(
    f"  Polygons with null Cluster_ID     : "
    f"{null_cluster_count:,}"
)

print(
    f"  Polygons skipped (null values)    : "
    f"{skipped_nulls:,}"
)

print(
    f"  Unique states found               : "
    f"{len(state_clusters):,}"
)

print(
    f"  States                            : "
    f"{sorted(state_clusters.keys())}"
)

total_clusters = sum(
    len(clusters)
    for clusters in state_clusters.values()
)

print(
    f"  Unique clusters across all states : "
    f"{total_clusters:,}"
)


# ── STEP 2: RANK UNIQUE CLUSTERS WITHIN EACH STATE ─────────────────────────────

print("=" * 60)
print("STEP 2 - RANKING UNIQUE CLUSTERS WITHIN EACH STATE")
print("=" * 60)

# cluster_rank[Cluster_ID] = rank
cluster_rank = {}


for state in sorted(state_clusters.keys()):

    clusters = state_clusters[state]

    # Sort by:
    #   1. Suitability descending
    #   2. Capacity descending

    sorted_clusters = sorted(
        clusters.items(),
        key=lambda x: (x[1][0], x[1][1]),
        reverse=True
    )

    print(
        f"  [{state}] "
        f"{len(sorted_clusters):,} unique clusters ranked"
    )

    for rank, (cluster_id, (suitability, capacity)) in enumerate(
        sorted_clusters,
        start=1
    ):

        cluster_rank[cluster_id] = rank

        # Print top 3 for each state
        if rank <= 3:

            print(
                f"    Rank {rank} -> "
                f"Cluster {cluster_id} | "
                f"Suitability: {suitability:.4f} | "
                f"Capacity: {capacity:,.2f} kWdc"
            )


print(
    f"Total unique clusters ranked : "
    f"{len(cluster_rank):,}"
)


# ── STEP 3: UPDATE Cluster_State_Rank ─────────────────────────────────────────

print("=" * 60)
print("STEP 3 - UPDATING Cluster_State_Rank FIELD")
print("=" * 60)

updated = 0
nulled = 0
errors = 0

log_interval = max(1, total_rows // 10)

print("Writing Cluster_State_Rank values...")


with arcpy.da.UpdateCursor(
    FEATURE_CLASS,
    ["Cluster_ID", "Cluster_State_Rank"]
) as cursor:

    for i, row in enumerate(cursor):

        cluster_id = row[0]

        try:

            if cluster_id is None:

                row[1] = None
                nulled += 1

            else:

                rank = cluster_rank.get(cluster_id)

                if rank is not None:
                    row[1] = rank
                    updated += 1
                else:
                    row[1] = None
                    nulled += 1

            cursor.updateRow(row)

        except Exception as e:

            print(
                f"  Error on row {i}: {e}"
            )

            errors += 1

        if i % log_interval == 0 and i > 0:

            pct = (i / total_rows) * 100

            print(
                f"  Update progress: "
                f"{i:,} / {total_rows:,} ({pct:.0f}%) | "
                f"Updated: {updated:,} | "
                f"Nulled: {nulled:,}"
            )


# ── COMPLETE ──────────────────────────────────────────────────────────────────

end_time = datetime.now()
elapsed = end_time - start_time

print("=" * 60)
print("COMPLETE")
print(f"Finished : {end_time.strftime('%Y-%m-%d %H:%M:%S')}")
print(
    f"Elapsed  : "
    f"{str(elapsed).split('.')[0]}"
)

print(
    f"Rows updated with rank value : "
    f"{updated:,}"
)

print(
    f"Rows set to null             : "
    f"{nulled:,}"
)

print(
    f"Errors                       : "
    f"{errors:,}"
)

print("=" * 60)
```
