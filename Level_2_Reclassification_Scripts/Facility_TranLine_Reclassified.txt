import arcpy

# Set your feature class path
fc = r"C:\Users\KyleSteen.AzureAD\Documents\ArcGIS\Projects\L2_Suitability\Level_2_Analysis_AEP_GlintGlare_LCOE_Shape_Size_Substations_TransLines_4_15_2026.gdb\Alaska_Level_2_Analysis_AEP_GlintGlare_LCOE_Shape_Size_Substations_TransLines"

field_name = "Facility_TransLine_Distance_Meters_Reclassified"

# Add new field if it doesn't exist
existing_fields = [f.name for f in arcpy.ListFields(fc)]
if field_name not in existing_fields:
    arcpy.management.AddField(fc, field_name, "DOUBLE")
    print(f"Added field: {field_name}")

# Reclassification logic
def reclassify(value):
    if value is None:
        return None
    elif value > 4000:
        return 1
    elif value >= 3000:
        return 4
    elif value >= 2000:
        return 7
    else:
        return 10

# Update the new field using a cursor
with arcpy.da.UpdateCursor(fc, ["Facility_TransLine_Distance_Meters", field_name]) as cursor:
    for row in cursor:
        row[1] = reclassify(row[0])
        cursor.updateRow(row)

print("Reclassification complete!")
