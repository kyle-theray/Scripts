import arcpy

fc = r"C:\Users\KyleSteen.AzureAD\Documents\ArcGIS\Projects\L2_Suitability\Level_2_Analysis_AEP_GlintGlare_LCOE_Shape_Size_Substations_TransLines_4_15_2026.gdb\Alaska_Level_2_Analysis_AEP_GlintGlare_LCOE_Shape_Size_Substations_TransLines"

field_name = "Facility_AEP_Mean_Reclassified"

existing_fields = [f.name for f in arcpy.ListFields(fc)]
if field_name not in existing_fields:
    arcpy.management.AddField(fc, field_name, "DOUBLE")
    print(f"Added field: {field_name}")

def reclassify_aep(value):
    if value is None:
        return None
    value = round(value, 4)
    if value >= 9557.55:
        return 10
    elif value >= 9248.34:
        return 9
    elif value >= 8798.57:
        return 8
    elif value >= 8376.91:
        return 7
    elif value >= 7955.26:
        return 6
    elif value >= 7561.71:
        return 5
    elif value >= 7168.16:
        return 4
    elif value >= 6887.06:
        return 3
    elif value >= 6549.73:
        return 2
    else:
        return 1

with arcpy.da.UpdateCursor(fc, ["Facility_AEP_Mean", field_name]) as cursor:
    for row in cursor:
        row[1] = reclassify_aep(row[0])
        cursor.updateRow(row)

print("AEP Reclassification complete!")
