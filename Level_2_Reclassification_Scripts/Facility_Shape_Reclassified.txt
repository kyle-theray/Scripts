import arcpy

# Set your feature class path
fc = r"C:\Users\KyleSteen.AzureAD\Documents\ArcGIS\Projects\L2_Suitability\Level_2_Analysis_AEP_GlintGlare_LCOE_Shape_Size_Substations_TransLines_4_15_2026.gdb\Alaska_Level_2_Analysis_AEP_GlintGlare_LCOE_Shape_Size_Substations_TransLines"  # e.g., r"C:\Project\MyGDB.gdb\MyFeatureClass"

field_name = "Facility_Shape_Reclassified"

# Add new field if it doesn't exist
existing_fields = [f.name for f in arcpy.ListFields(fc)]
if field_name not in existing_fields:
    arcpy.management.AddField(fc, field_name, "DOUBLE")
    print(f"Added field: {field_name}")

# Reclassification logic
def reclassify(value):
    if value is None:
        return None
    elif value < 0.2:
        return 2
    elif value <= 0.6:
        return 5
    elif value <= 0.8:
        return 7
    else:
        return 9

# Update the new field using a cursor
with arcpy.da.UpdateCursor(fc, ["Facility_Shape", field_name]) as cursor:
    for row in cursor:
        row[1] = reclassify(row[0])
        cursor.updateRow(row)

print("Reclassification complete!")
