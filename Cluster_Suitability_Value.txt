"""
calculate_cluster_suitability.py

PURPOSE:
    For each cluster in the ROW feature class, calculate a capacity-weighted
    average Suitability_Value and assign it to every polygon in that cluster
    under the Cluster_Suitability_Value column.
    Polygons not in a cluster receive null.

FORMULA:
    Cluster_Suitability_Value = SUM( (Capacity_kWdc_i / Cluster_Capacity_kWdc)
                                      * Suitability_Value_i )

    Where:
        - Capacity_kWdc_i       : Individual polygon capacity
        - Cluster_Capacity_kWdc : Total capacity of the cluster
        - Suitability_Value_i   : Individual polygon suitability score

INPUTS:
    - Feature class path (set in configuration below)

OUTPUTS:
    - Cluster_Suitability_Value field updated in place

NOTES:
    - Run in ArcGIS Pro notebook
    - Requires arcpy (available by default in ArcGIS Pro)
    - Cluster_Capacity_kWdc must already be populated (run calculate_cluster_capacity.py first)
    - Polygons with null Cluster_ID receive null Cluster_Suitability_Value
"""

import arcpy
from collections import defaultdict
from datetime import datetime

# ── USER CONFIGURATION ────────────────────────────────────────────────────────
FEATURE_CLASS = r"C:\Users\KyleSteen.AzureAD\Documents\ArcGIS\Projects\Hawaii_ROW_comparison\Level_1_Analysis_Hawaii_Clustering.gdb\Level_1_Analysis_Hawaii_Clustering"  # <-- UPDATE
# ─────────────────────────────────────────────────────────────────────────────

start_time = datetime.now()
print("=" * 60)
print("CALCULATE CLUSTER SUITABILITY VALUE")
print(f"Started : {start_time.strftime('%Y-%m-%d %H:%M:%S')}")
print("=" * 60)

# --- Validate feature class exists ---
print(f"Feature class : {FEATURE_CLASS}")
if not arcpy.Exists(FEATURE_CLASS):
    raise FileNotFoundError(f"Feature class not found: {FEATURE_CLASS}")
print("Feature class confirmed.")

# --- Get total row count ---
total_rows = int(arcpy.GetCount_management(FEATURE_CLASS).getOutput(0))
print(f"Total rows    : {total_rows:,}")

# ── STEP 1: BUILD WEIGHTED SUITABILITY LOOKUP ─────────────────────────────────
print("=" * 60)
print("STEP 1 - BUILDING CAPACITY-WEIGHTED SUITABILITY LOOKUP")
print("=" * 60)

# For each cluster accumulate the weighted suitability sum
# weighted_suitability[cluster_id] = SUM( (capacity_i / cluster_total) * suitability_i )
# We read Cluster_Capacity_kWdc directly from the attribute table since it was
# already calculated and stored by calculate_cluster_capacity.py

cluster_weighted_suitability = defaultdict(float)
cluster_counts               = defaultdict(int)
null_cluster_count           = 0
skipped_null_capacity        = 0
skipped_null_suitability     = 0
skipped_zero_cluster_cap     = 0

fields = ["Cluster_ID", "Capacity_kWdc", "Cluster_Capacity_kWdc", "Suitability_Value"]

print(f"Reading fields: {fields}")
print("Calculating capacity-weighted suitability per cluster...")

with arcpy.da.SearchCursor(FEATURE_CLASS, fields) as cursor:
    for row in cursor:
        cluster_id        = row[0]
        capacity          = row[1]
        cluster_capacity  = row[2]
        suitability       = row[3]

        # --- Skip polygons not in a cluster ---
        if cluster_id is None:
            null_cluster_count += 1
            continue

        # --- Skip if any required value is null ---
        if capacity is None:
            skipped_null_capacity += 1
            continue

        if suitability is None:
            skipped_null_suitability += 1
            continue

        if cluster_capacity is None or cluster_capacity == 0:
            skipped_zero_cluster_cap += 1
            continue

        # --- Calculate weight and add weighted suitability ---
        weight    = capacity / cluster_capacity
        weighted  = weight * suitability

        cluster_weighted_suitability[cluster_id] += weighted
        cluster_counts[cluster_id]               += 1

print("Read complete.")
print(f"  Polygons with null Cluster_ID              : {null_cluster_count:,}")
print(f"  Polygons skipped (null Capacity_kWdc)      : {skipped_null_capacity:,}")
print(f"  Polygons skipped (null Suitability_Value)  : {skipped_null_suitability:,}")
print(f"  Polygons skipped (null/zero cluster cap)   : {skipped_zero_cluster_cap:,}")
print(f"  Unique clusters calculated                 : {len(cluster_weighted_suitability):,}")
print(f"  Total polygons included in calculation     : {sum(cluster_counts.values()):,}")

# --- Log top 5 clusters by weighted suitability ---
if cluster_weighted_suitability:
    print("Top 5 clusters by Cluster_Suitability_Value:")
    top5 = sorted(cluster_weighted_suitability.items(), key=lambda x: x[1], reverse=True)[:5]
    for cluster_id, wsv in top5:
        print(f"  Cluster {cluster_id} : {wsv:.4f} ({cluster_counts[cluster_id]:,} polygons)")

    print("Bottom 5 clusters by Cluster_Suitability_Value:")
    bottom5 = sorted(cluster_weighted_suitability.items(), key=lambda x: x[1])[:5]
    for cluster_id, wsv in bottom5:
        print(f"  Cluster {cluster_id} : {wsv:.4f} ({cluster_counts[cluster_id]:,} polygons)")

# ── STEP 2: UPDATE Cluster_Suitability_Value FIELD ───────────────────────────
print("=" * 60)
print("STEP 2 - UPDATING Cluster_Suitability_Value FIELD")
print("=" * 60)

updated      = 0
nulled       = 0
errors       = 0
log_interval = max(1, total_rows // 10)

print("Writing Cluster_Suitability_Value values...")
with arcpy.da.UpdateCursor(FEATURE_CLASS, ["Cluster_ID", "Cluster_Suitability_Value"]) as cursor:
    for i, row in enumerate(cursor):
        cluster_id = row[0]

        try:
            if cluster_id is None:
                row[1] = None
                nulled += 1
            else:
                row[1] = cluster_weighted_suitability.get(cluster_id, None)
                updated += 1

            cursor.updateRow(row)

        except Exception as e:
            print(f"  Error on row {i}: {e}")
            errors += 1

        if i % log_interval == 0 and i > 0:
            pct = (i / total_rows) * 100
            print(f"  Update progress: {i:,} / {total_rows:,} ({pct:.0f}%) | Updated: {updated:,} | Nulled: {nulled:,}")

# ── COMPLETE ──────────────────────────────────────────────────────────────────
end_time = datetime.now()
elapsed  = end_time - start_time

print("=" * 60)
print("COMPLETE")
print(f"Finished : {end_time.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"Elapsed  : {str(elapsed).split('.')[0]}")
print(f"Rows updated with suitability value : {updated:,}")
print(f"Rows set to null                    : {nulled:,}")
print(f"Errors                              : {errors:,}")
print("=" * 60)
