"""
calculate_state_rank.py

PURPOSE:
    Rank all polygons within each state by Suitability_Value descending.
    Tiebreaker is Capacity_kWdc descending.

RANKING LOGIC:
    - Grouped by State_Name
    - Primary sort   : Suitability_Value descending
    - Tiebreaker     : Capacity_kWdc descending
    - Rank 1 = highest suitability polygon in that state

OUTPUT:
    - State_Rank field updated in place

NOTES:
    - Run in an ArcGIS Pro notebook
    - Requires arcpy
    - Polygons with null State_Name, Suitability_Value, or Capacity_kWdc
      are skipped and their State_Rank is set to null
"""

import arcpy
from collections import defaultdict
from datetime import datetime


# ── USER CONFIGURATION ────────────────────────────────────────────────────────
FEATURE_CLASS = r"path\to\your\geodatabase.gdb\your_feature_class"  # <-- UPDATE
# ─────────────────────────────────────────────────────────────────────────────


start_time = datetime.now()

print("=" * 60)
print("CALCULATE STATE RANK")
print(f"Started : {start_time.strftime('%Y-%m-%d %H:%M:%S')}")
print("=" * 60)


# ── VALIDATE FEATURE CLASS ────────────────────────────────────────────────────

print(f"Feature class : {FEATURE_CLASS}")

if not arcpy.Exists(FEATURE_CLASS):
    raise FileNotFoundError(f"Feature class not found: {FEATURE_CLASS}")

print("Feature class confirmed.")


# ── GET TOTAL ROW COUNT ───────────────────────────────────────────────────────

total_rows = int(arcpy.GetCount_management(FEATURE_CLASS).getOutput(0))
print(f"Total rows    : {total_rows:,}")


# ── STEP 1: READ VALUES AND GROUP BY STATE ────────────────────────────────────

print("=" * 60)
print("STEP 1 - READING VALUES PER STATE")
print("=" * 60)

# state_polygons[state] = [(OID, suitability, capacity), ...]
state_polygons = defaultdict(list)

skipped_nulls = 0

fields = [
    "OID@",
    "State_Name",
    "Suitability_Value",
    "Capacity_kWdc"
]

print(f"Reading fields: {fields}")

with arcpy.da.SearchCursor(FEATURE_CLASS, fields) as cursor:

    for row in cursor:

        oid         = row[0]
        state_name  = row[1]
        suitability = row[2]
        capacity    = row[3]

        # Skip polygons with missing ranking information
        if state_name is None or suitability is None or capacity is None:
            skipped_nulls += 1
            continue

        state_polygons[state_name].append(
            (oid, suitability, capacity)
        )


print("Read complete.")
print(f"  Polygons skipped (null values) : {skipped_nulls:,}")
print(f"  Unique states found            : {len(state_polygons):,}")
print(f"  States                         : {sorted(state_polygons.keys())}")

total_polygons_ranked = sum(
    len(polygons) for polygons in state_polygons.values()
)

print(f"  Total polygons to rank         : {total_polygons_ranked:,}")


# ── STEP 2: RANK POLYGONS WITHIN EACH STATE ───────────────────────────────────

print("=" * 60)
print("STEP 2 - RANKING POLYGONS WITHIN EACH STATE")
print("=" * 60)

# rank_lookup[OID] = rank
rank_lookup = {}

for state in sorted(state_polygons.keys()):

    polygons = state_polygons[state]

    # Sort:
    #   1. Suitability_Value descending
    #   2. Capacity_kWdc descending
    sorted_polygons = sorted(
        polygons,
        key=lambda x: (x[1], x[2]),
        reverse=True
    )

    print(f"  [{state}] {len(sorted_polygons):,} polygons ranked")

    for rank, (oid, suitability, capacity) in enumerate(
        sorted_polygons,
        start=1
    ):

        rank_lookup[oid] = rank

        # Log top 3 per state
        if rank <= 3:
            print(
                f"    Rank {rank} -> OID {oid} | "
                f"Suitability: {suitability:.4f} | "
                f"Capacity: {capacity:,.2f} kWdc"
            )


print(f"Total polygons ranked : {len(rank_lookup):,}")


# ── STEP 3: UPDATE STATE_RANK FIELD ───────────────────────────────────────────

print("=" * 60)
print("STEP 3 - UPDATING State_Rank FIELD")
print("=" * 60)

updated = 0
nulled = 0
errors = 0

log_interval = max(1, total_rows // 10)

print("Writing State_Rank values...")

with arcpy.da.UpdateCursor(
    FEATURE_CLASS,
    ["OID@", "State_Rank"]
) as cursor:

    for i, row in enumerate(cursor):

        oid = row[0]

        try:

            if oid in rank_lookup:
                row[1] = rank_lookup[oid]
                updated += 1

            else:
                row[1] = None
                nulled += 1

            cursor.updateRow(row)

        except Exception as e:

            print(f"  Error on OID {oid}: {e}")
            errors += 1

        if i % log_interval == 0 and i > 0:

            pct = (i / total_rows) * 100

            print(
                f"  Update progress: "
                f"{i:,} / {total_rows:,} ({pct:.0f}%) | "
                f"Updated: {updated:,} | "
                f"Nulled: {nulled:,}"
            )


# ── COMPLETE ──────────────────────────────────────────────────────────────────

end_time = datetime.now()
elapsed = end_time - start_time

print("=" * 60)
print("COMPLETE")
print(f"Finished : {end_time.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"Elapsed  : {str(elapsed).split('.')[0]}")
print(f"Rows updated with rank value : {updated:,}")
print(f"Rows set to null             : {nulled:,}")
print(f"Errors                       : {errors:,}")
print("=" * 60)
