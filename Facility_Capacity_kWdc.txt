"""
calculate_facility_capacity.py

PURPOSE:
    Populates Facility_Capacity_kWdc for every row in the feature class.
    Clustered rows copy from Cluster_Capacity_kWdc.
    Non-clustered rows copy from Capacity_kWdc.

LOGIC:
    - Cluster_ID is not null -> Facility_Capacity_kWdc = Cluster_Capacity_kWdc
    - Cluster_ID is null     -> Facility_Capacity_kWdc = Capacity_kWdc

INPUTS:
    - Feature class path (set in configuration below)

OUTPUTS:
    - Facility_Capacity_kWdc field updated in place

NOTES:
    - Run in ArcGIS Pro notebook
    - Requires arcpy (available by default in ArcGIS Pro)
    - Cluster_Capacity_kWdc must already be populated before running this script
"""

import arcpy
from datetime import datetime

# ── USER CONFIGURATION ────────────────────────────────────────────────────────
FEATURE_CLASS = r"C:\Users\KyleSteen.AzureAD\Documents\ArcGIS\Projects\L1_clustering\Level_1_Analysis_Clustering.gdb\Level_1_Analysis_CONUS_Clustering"  # <-- UPDATE
# ─────────────────────────────────────────────────────────────────────────────

start_time = datetime.now()
print("=" * 60)
print("CALCULATE FACILITY CAPACITY kWdc")
print(f"Started : {start_time.strftime('%Y-%m-%d %H:%M:%S')}")
print("=" * 60)

# --- Validate feature class exists ---
print(f"Feature class : {FEATURE_CLASS}")
if not arcpy.Exists(FEATURE_CLASS):
    raise FileNotFoundError(f"Feature class not found: {FEATURE_CLASS}")
print("Feature class confirmed.")

# --- Get total row count ---
total_rows = int(arcpy.GetCount_management(FEATURE_CLASS).getOutput(0))
print(f"Total rows    : {total_rows:,}")

# ── UPDATE FACILITY_CAPACITY_kWdc FIELD ───────────────────────────────────────
print("=" * 60)
print("UPDATING Facility_Capacity_kWdc FIELD")
print("=" * 60)

fields = ["Cluster_ID", "Capacity_kWdc", "Cluster_Capacity_kWdc", "Facility_Capacity_kWdc"]

clustered     = 0
non_clustered = 0
errors        = 0
null_warnings = 0
log_interval  = max(1, total_rows // 10)

print("Writing Facility_Capacity_kWdc values...")
with arcpy.da.UpdateCursor(FEATURE_CLASS, fields) as cursor:
    for i, row in enumerate(cursor):
        cluster_id       = row[0]
        capacity         = row[1]
        cluster_capacity = row[2]

        try:
            if cluster_id is not None:
                # Clustered row — copy from Cluster_Capacity_kWdc
                if cluster_capacity is None:
                    print(f"  Warning: Cluster_Capacity_kWdc is null for Cluster_ID {cluster_id} on row {i}")
                    null_warnings += 1
                row[3]     = cluster_capacity
                clustered += 1
            else:
                # Non-clustered row — copy from Capacity_kWdc
                if capacity is None:
                    print(f"  Warning: Capacity_kWdc is null for non-clustered row {i}")
                    null_warnings += 1
                row[3]         = capacity
                non_clustered += 1

            cursor.updateRow(row)

        except Exception as e:
            print(f"  Error on row {i}: {e}")
            errors += 1

        if i % log_interval == 0 and i > 0:
            pct = (i / total_rows) * 100
            print(f"  Update progress: {i:,} / {total_rows:,} ({pct:.0f}%) | Clustered: {clustered:,} | Non-clustered: {non_clustered:,}")

# ── COMPLETE ──────────────────────────────────────────────────────────────────
end_time = datetime.now()
elapsed  = end_time - start_time

print("=" * 60)
print("COMPLETE")
print(f"Finished                                    : {end_time.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"Elapsed                                     : {str(elapsed).split('.')[0]}")
print(f"Clustered rows (copied Cluster_Capacity)    : {clustered:,}")
print(f"Non-clustered rows (copied Capacity_kWdc)   : {non_clustered:,}")
print(f"Null value warnings                         : {null_warnings:,}")
print(f"Errors                                      : {errors:,}")
print("=" * 60)
