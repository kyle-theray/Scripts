// ======================================================================
// CONUS ROW Slope & Aspect Batch Export Script
// ======================================================================

// --------------------------------------
// INPUTS
// --------------------------------------
var polygonsAsset = 'projects/ee-kylesteen/assets/CONUS_3_3';
var exportFolder  = 'CONUS_Chunks';   // inside GEE_Exports in Google Drive
var exportScale   = 10;                // DEM resolution in meters
var bufferDistance = 50;               // 5x5 pixel neighborhood support
var targetCRS     = 'EPSG:5070';       // CONUS Albers
var batchSize     = 50000;             // number of polygons per batch
var flatThreshold = 0.87;              // Slope percent rise threshold to consider "flat"

// --------------------------------------
// LOAD AND PREPARE POLYGONS
// --------------------------------------
var polygons = ee.FeatureCollection(polygonsAsset)
  .select(['ROW_ID', 'Square_Met']);   // keep your area column

// Reproject polygons for computation (planar CRS)
var polygonsProj = polygons.map(function(f){
  return f.transform(targetCRS, 1);
});

// Buffer polygons for slope/aspect neighborhood support only
var polygonsBuffer = polygonsProj.map(function(f){
  return f.buffer(bufferDistance);
});

// --------------------------------------
// LOAD DEM (USGS 3DEP 10m) & REPROJECT
// --------------------------------------
var dem = ee.ImageCollection('USGS/3DEP/10m_collection').mosaic().toFloat();
var demProj = dem.reproject({crs: targetCRS, scale: exportScale});

// Clip DEM to buffered polygons
var demClipped = demProj.clip(polygonsBuffer);

// --------------------------------------
// COMPUTE SLOPE AND ASPECT
// --------------------------------------
var slopeDeg  = ee.Terrain.slope(demClipped).rename('Slope_deg');
var slopePerc = slopeDeg
   .multiply(Math.PI / 180)
   .tan()
   .multiply(100)
   .rename('Slope_per');

var aspectDeg = ee.Terrain.aspect(demClipped).rename('Aspect_deg');

// Convert aspect to 8 cardinal bins (0–7)
var aspectCat = aspectDeg
   .divide(45)
   .add(0.5)
   .floor()
   .mod(8)
   .rename('Aspect_cat');

// --------------------------------------
// VISUALIZATION PARAMETERS
// --------------------------------------

// Slope: low = green, medium = yellow, high = red
var slopePercVis  = {min:0, max:60, palette:['green','yellow','red']};
var slopeDegVis   = {min:0, max:60, palette:['green','yellow','red']};

// Aspect: South/flat darkest green, SW/SE lighter green, E/W yellow, NE/NW red-yellow, N red
var aspectCatVis  = {
  min:0,
  max:7,
  palette:[
    'red',          // 0 = N
    'orange',       // 1 = NE
    'yellow',       // 2 = E
    'lightgreen',   // 3 = SE
    'darkgreen',    // 4 = S
    'lightgreen',   // 5 = SW
    'yellow',       // 6 = W
    'orange'        // 7 = NW
  ]
};

var demVis = {min:0, max:4000, palette:['white','gray','black']};

// --------------------------------------
// Add layers to map
// --------------------------------------
Map.addLayer(demClipped, demVis, 'DEM (buffered)', false);
Map.addLayer(slopeDeg, slopeDegVis, 'Slope (deg)', false);
Map.addLayer(slopePerc, slopePercVis, 'Slope (%)', false);
Map.addLayer(aspectDeg, {min:0,max:360,palette:['blue','cyan','green','yellow','orange','red','magenta','purple','blue']}, 'Aspect (deg)', false);
Map.addLayer(aspectCat, aspectCatVis, 'Aspect categorical (0–7)', false);
Map.addLayer(polygonsProj, {color:'red'}, 'Original polygons', true);
Map.addLayer(polygonsBuffer, {color:'blue', fillColor:'00000000'}, 'Buffer 50m', false);

// --------------------------------------
// FUNCTION TO COMPUTE ZONAL STATS WITH FLAT-AREA ASPECT
// --------------------------------------
var computeStats = function(f) {
  var geom = f.geometry();

  // Mean slope (%)
  var slope_mean = slopePerc.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: geom,
    scale: exportScale,
    maxPixels: 1e13,
    tileScale: 4
  }).get('Slope_per');

  // Mode aspect (categorical 0–7)
  var aspect_mode = aspectCat.reduceRegion({
    reducer: ee.Reducer.mode(),
    geometry: geom,
    scale: exportScale,
    maxPixels: 1e13,
    tileScale: 4
  }).get('Aspect_cat');

  // Apply flat terrain rule: slope ≤ flatThreshold → aspect = -1
  aspect_mode = ee.Algorithms.If(
    ee.Number(slope_mean).lte(flatThreshold),
    -1,
    aspect_mode
  );

  return f.set({
    'Slope_Mean_Percentage': slope_mean,
    'Aspect_Mode_Numeric': aspect_mode
  });
};

// --------------------------------------
// BATCH EXPORTING
// --------------------------------------
var numPolygons = polygonsProj.size();
var numBatches = numPolygons.divide(batchSize).ceil();

print('Total polygons:', numPolygons);
print('Number of batches:', numBatches);

// Create a list of batch indices
var batchIndices = ee.List.sequence(0, numBatches.subtract(1));

batchIndices.getInfo().forEach(function(i){
  var start = i * batchSize;
  var batchList = polygonsProj.toList(batchSize, start);
  var batchFC = ee.FeatureCollection(batchList).map(computeStats);

  // Export CSV per batch including Square_Met
  Export.table.toDrive({
    collection: batchFC,
    description: 'CONUS_ROW_Slope_Aspect_batch_' + i,
    folder: exportFolder,
    fileFormat: 'CSV',
    selectors: ['ROW_ID', 'Square_Met', 'Aspect_Mode_Numeric', 'Slope_Mean_Percentage']
  });

  print('Submitted batch:', i + 1, 'of', numBatches.getInfo());
});

// ======================================================================
// SCIENTIFIC & METHODOLOGICAL NOTES
// ======================================================================
/*
1. DEM Source
   - USGS 3DEP 10 m DEM
2. CRS Handling
   - EPSG:5070 (CONUS Albers)
3. Buffer Logic
   - 50 m buffer for slope/aspect support
4. Flat Terrain Rule
   - Slope ≤ 0.87% considered flat → aspect = -1
5. Aspect Statistics
   - 8 cardinal classes: 0=N, 1=NE, 2=E, 3=SE, 4=S, 5=SW, 6=W, 7=NW
6. Slope Statistics
   - Percent rise = tan(slope_deg) × 100
7. Batch Processing
   - Batch export prevents timeouts
8. Visualization
   - Updated slope: green → yellow → red
   - Updated aspect: N=red, NE/NW=orange, E/W=yellow, SE/SW=light green, S/dark green
*/
