"""
calculate_facility_id.py

PURPOSE:
    Assigns a unique Facility_ID to every row in the feature class.
    Rows in the same cluster share the same Facility_ID.
    Rows not in a cluster get their own unique Facility_ID.

SEQUENCING:
    - Clustered rows    : Facility_ID 1 to N (one per unique cluster)
    - Non-clustered rows: Facility_ID N+1 onwards (one per individual row)

INPUTS:
    - Feature class path (set in configuration below)

OUTPUTS:
    - Facility_ID field updated in place

NOTES:
    - Run in ArcGIS Pro notebook
    - Requires arcpy (available by default in ArcGIS Pro)
    - Cluster_ID must already be populated before running this script
"""

import arcpy
from datetime import datetime

# ── USER CONFIGURATION ────────────────────────────────────────────────────────
FEATURE_CLASS = r"C:\Users\KyleSteen.AzureAD\Documents\ArcGIS\Projects\L1_clustering\Level_1_Analysis_Clustering.gdb\Level_1_Analysis_CONUS_Clustering"  # <-- UPDATE
# ─────────────────────────────────────────────────────────────────────────────

start_time = datetime.now()
print("=" * 60)
print("CALCULATE FACILITY ID")
print(f"Started : {start_time.strftime('%Y-%m-%d %H:%M:%S')}")
print("=" * 60)

# --- Validate feature class exists ---
print(f"Feature class : {FEATURE_CLASS}")
if not arcpy.Exists(FEATURE_CLASS):
    raise FileNotFoundError(f"Feature class not found: {FEATURE_CLASS}")
print("Feature class confirmed.")

# --- Get total row count ---
total_rows = int(arcpy.GetCount_management(FEATURE_CLASS).getOutput(0))
print(f"Total rows    : {total_rows:,}")

# ── STEP 1: BUILD CLUSTER TO FACILITY_ID LOOKUP ───────────────────────────────
print("=" * 60)
print("STEP 1 - ASSIGNING FACILITY IDs TO CLUSTERS")
print("=" * 60)

# Assign one sequential Facility_ID per unique Cluster_ID
cluster_to_facility = {}
facility_counter    = 1
null_cluster_count  = 0

print("Reading Cluster_ID values...")
with arcpy.da.SearchCursor(FEATURE_CLASS, ["Cluster_ID"]) as cursor:
    for row in cursor:
        cluster_id = row[0]

        if cluster_id is None:
            null_cluster_count += 1
            continue

        if cluster_id not in cluster_to_facility:
            cluster_to_facility[cluster_id] = facility_counter
            facility_counter += 1

clustered_facilities   = facility_counter - 1
print(f"Read complete.")
print(f"  Unique clusters found               : {clustered_facilities:,}")
print(f"  Facility IDs assigned to clusters   : 1 to {clustered_facilities:,}")
print(f"  Polygons with null Cluster_ID       : {null_cluster_count:,}")
print(f"  Non-clustered IDs will start at     : {facility_counter:,}")

# ── STEP 2: UPDATE FACILITY_ID FIELD ─────────────────────────────────────────
print("=" * 60)
print("STEP 2 - UPDATING FACILITY_ID FIELD")
print("=" * 60)

updated      = 0
isolated     = 0
errors       = 0
log_interval = max(1, total_rows // 10)

print("Writing Facility_ID values...")
with arcpy.da.UpdateCursor(FEATURE_CLASS, ["Cluster_ID", "Facility_ID"]) as cursor:
    for i, row in enumerate(cursor):
        cluster_id = row[0]

        try:
            if cluster_id is None:
                # Non-clustered row gets next sequential ID
                row[1]           = facility_counter
                facility_counter += 1
                isolated         += 1
            else:
                # Clustered row gets shared cluster Facility_ID
                row[1]   = cluster_to_facility[cluster_id]
                updated += 1

            cursor.updateRow(row)

        except Exception as e:
            print(f"  Error on row {i}: {e}")
            errors += 1

        if i % log_interval == 0 and i > 0:
            pct = (i / total_rows) * 100
            print(f"  Update progress: {i:,} / {total_rows:,} ({pct:.0f}%) | Clustered: {updated:,} | Isolated: {isolated:,}")

# ── COMPLETE ──────────────────────────────────────────────────────────────────
end_time      = datetime.now()
elapsed       = end_time - start_time
total_assigned = updated + isolated

print("=" * 60)
print("COMPLETE")
print(f"Finished                              : {end_time.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"Elapsed                               : {str(elapsed).split('.')[0]}")
print(f"Total Facility_IDs assigned           : {total_assigned:,}")
print(f"  Clustered rows (shared IDs)         : {updated:,}")
print(f"  Non-clustered rows (unique IDs)     : {isolated:,}")
print(f"  Cluster Facility_ID range           : 1 to {clustered_facilities:,}")
print(f"  Isolated Facility_ID range          : {clustered_facilities + 1:,} to {total_assigned:,}")
print(f"Errors                                : {errors:,}")
print("=" * 60)
