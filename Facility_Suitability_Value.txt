"""
calculate_facility_suitability.py

PURPOSE:
    Populates Facility_Suitability_Value for every row in the feature class.
    Clustered rows copy from Cluster_Suitability_Value.
    Non-clustered rows copy from Suitability_Value.

LOGIC:
    - Cluster_ID is not null -> Facility_Suitability_Value = Cluster_Suitability_Value
    - Cluster_ID is null     -> Facility_Suitability_Value = Suitability_Value

INPUTS:
    - Feature class path (set in configuration below)

OUTPUTS:
    - Facility_Suitability_Value field updated in place

NOTES:
    - Run in ArcGIS Pro notebook
    - Requires arcpy (available by default in ArcGIS Pro)
    - Cluster_Suitability_Value must already be populated before running this script
"""

import arcpy
from datetime import datetime

# ── USER CONFIGURATION ────────────────────────────────────────────────────────
FEATURE_CLASS = r"C:\Users\KyleSteen.AzureAD\Documents\ArcGIS\Projects\L1_clustering\Level_1_Analysis_Clustering.gdb\Level_1_Analysis_CONUS_Clustering"  # <-- UPDATE
# ─────────────────────────────────────────────────────────────────────────────

start_time = datetime.now()
print("=" * 60)
print("CALCULATE FACILITY SUITABILITY VALUE")
print(f"Started : {start_time.strftime('%Y-%m-%d %H:%M:%S')}")
print("=" * 60)

# --- Validate feature class exists ---
print(f"Feature class : {FEATURE_CLASS}")
if not arcpy.Exists(FEATURE_CLASS):
    raise FileNotFoundError(f"Feature class not found: {FEATURE_CLASS}")
print("Feature class confirmed.")

# --- Get total row count ---
total_rows = int(arcpy.GetCount_management(FEATURE_CLASS).getOutput(0))
print(f"Total rows    : {total_rows:,}")

# ── UPDATE FACILITY_SUITABILITY_VALUE FIELD ───────────────────────────────────
print("=" * 60)
print("UPDATING Facility_Suitability_Value FIELD")
print("=" * 60)

fields = ["Cluster_ID", "Suitability_Value", "Cluster_Suitability_Value", "Facility_Suitability_Value"]

clustered     = 0
non_clustered = 0
errors        = 0
null_warnings = 0
log_interval  = max(1, total_rows // 10)

print("Writing Facility_Suitability_Value values...")
with arcpy.da.UpdateCursor(FEATURE_CLASS, fields) as cursor:
    for i, row in enumerate(cursor):
        cluster_id           = row[0]
        suitability          = row[1]
        cluster_suitability  = row[2]

        try:
            if cluster_id is not None:
                # Clustered row — copy from Cluster_Suitability_Value
                if cluster_suitability is None:
                    print(f"  Warning: Cluster_Suitability_Value is null for Cluster_ID {cluster_id} on row {i}")
                    null_warnings += 1
                row[3]     = cluster_suitability
                clustered += 1
            else:
                # Non-clustered row — copy from Suitability_Value
                if suitability is None:
                    print(f"  Warning: Suitability_Value is null for non-clustered row {i}")
                    null_warnings += 1
                row[3]         = suitability
                non_clustered += 1

            cursor.updateRow(row)

        except Exception as e:
            print(f"  Error on row {i}: {e}")
            errors += 1

        if i % log_interval == 0 and i > 0:
            pct = (i / total_rows) * 100
            print(f"  Update progress: {i:,} / {total_rows:,} ({pct:.0f}%) | Clustered: {clustered:,} | Non-clustered: {non_clustered:,}")

# ── COMPLETE ──────────────────────────────────────────────────────────────────
end_time = datetime.now()
elapsed  = end_time - start_time

print("=" * 60)
print("COMPLETE")
print(f"Finished                                         : {end_time.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"Elapsed                                          : {str(elapsed).split('.')[0]}")
print(f"Clustered rows (copied Cluster_Suitability)      : {clustered:,}")
print(f"Non-clustered rows (copied Suitability_Value)    : {non_clustered:,}")
print(f"Null value warnings                              : {null_warnings:,}")
print(f"Errors                                           : {errors:,}")
print("=" * 60)
