"""
calculate_cluster_capacity.py

PURPOSE:
    For each cluster in the ROW feature class, sum all individual
    Capacity_kWdc values and assign the total to every polygon in
    that cluster under the Cluster_Capacity_kWdc column.
    Polygons not in a cluster receive null.

INPUTS:
    - Feature class path (set in configuration below)

OUTPUTS:
    - Cluster_Capacity_kWdc field updated in place

NOTES:
    - Run in ArcGIS Pro notebook
    - Requires arcpy (available by default in ArcGIS Pro)
    - Polygons with null Cluster_ID receive null Cluster_Capacity_kWdc
"""

import arcpy
from collections import defaultdict
from datetime import datetime

# ── USER CONFIGURATION ────────────────────────────────────────────────────────
FEATURE_CLASS = r"C:\Users\KyleSteen.AzureAD\Documents\ArcGIS\Projects\Hawaii_ROW_comparison\Level_1_Analysis_Hawaii_Clustering.gdb\Level_1_Analysis_Hawaii_Clustering"  # <-- UPDATE
# ─────────────────────────────────────────────────────────────────────────────

start_time = datetime.now()
print("=" * 60)
print("CALCULATE CLUSTER CAPACITY kWdc")
print(f"Started : {start_time.strftime('%Y-%m-%d %H:%M:%S')}")
print("=" * 60)

# --- Validate feature class exists ---
print(f"Feature class : {FEATURE_CLASS}")
if not arcpy.Exists(FEATURE_CLASS):
    raise FileNotFoundError(f"Feature class not found: {FEATURE_CLASS}")
print("Feature class confirmed.")

# --- Get total row count ---
total_rows = int(arcpy.GetCount_management(FEATURE_CLASS).getOutput(0))
print(f"Total rows    : {total_rows:,}")

# ── STEP 1: BUILD CLUSTER CAPACITY LOOKUP ────────────────────────────────────
print("=" * 60)
print("STEP 1 - BUILDING CLUSTER CAPACITY LOOKUP")
print("=" * 60)

# Sum Capacity_kWdc per Cluster_ID
cluster_capacity = defaultdict(float)
cluster_counts   = defaultdict(int)
null_count       = 0
no_capacity      = 0

print("Reading Cluster_ID and Capacity_kWdc values...")
with arcpy.da.SearchCursor(FEATURE_CLASS, ["Cluster_ID", "Capacity_kWdc"]) as cursor:
    for row in cursor:
        cluster_id   = row[0]
        capacity_val = row[1]

        if cluster_id is None:
            null_count += 1
            continue

        if capacity_val is None:
            no_capacity += 1
            continue

        cluster_capacity[cluster_id] += capacity_val
        cluster_counts[cluster_id]   += 1

print(f"Read complete.")
print(f"  Polygons with null Cluster_ID       : {null_count:,}")
print(f"  Polygons with null Capacity_kWdc    : {no_capacity:,}")
print(f"  Unique clusters found               : {len(cluster_capacity):,}")
print(f"  Total polygons in valid clusters    : {sum(cluster_counts.values()):,}")

# --- Log top 5 clusters by capacity ---
print("Top 5 clusters by total Cluster_Capacity_kWdc:")
top5 = sorted(cluster_capacity.items(), key=lambda x: x[1], reverse=True)[:5]
for cluster_id, total_cap in top5:
    print(f"  Cluster {cluster_id} : {total_cap:,.2f} kWdc ({cluster_counts[cluster_id]:,} polygons)")

# ── STEP 2: UPDATE CLUSTER_CAPACITY_kWdc ─────────────────────────────────────
print("=" * 60)
print("STEP 2 - UPDATING Cluster_Capacity_kWdc FIELD")
print("=" * 60)

updated   = 0
nulled    = 0
errors    = 0
log_interval = max(1, total_rows // 10)

print("Writing Cluster_Capacity_kWdc values...")
with arcpy.da.UpdateCursor(FEATURE_CLASS, ["Cluster_ID", "Cluster_Capacity_kWdc"]) as cursor:
    for i, row in enumerate(cursor):
        cluster_id = row[0]

        try:
            if cluster_id is None:
                row[1] = None
                nulled += 1
            else:
                row[1] = cluster_capacity.get(cluster_id, None)
                updated += 1

            cursor.updateRow(row)

        except Exception as e:
            print(f"  Error on row {i}: {e}")
            errors += 1

        if i % log_interval == 0 and i > 0:
            pct = (i / total_rows) * 100
            print(f"  Update progress: {i:,} / {total_rows:,} ({pct:.0f}%) | Updated: {updated:,} | Nulled: {nulled:,}")

# ── COMPLETE ──────────────────────────────────────────────────────────────────
end_time = datetime.now()
elapsed  = end_time - start_time

print("=" * 60)
print("COMPLETE")
print(f"Finished : {end_time.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"Elapsed  : {str(elapsed).split('.')[0]}")
print(f"Rows updated with capacity value : {updated:,}")
print(f"Rows set to null                 : {nulled:,}")
print(f"Errors                           : {errors:,}")
print("=" * 60)
